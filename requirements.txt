const $ = s => document.querySelector(s);

let state = {};            // id -> {status, note, sentAt}

const LS = "dentist-outreach-v1";

/* ---------- storage ---------- */
function lsRead(){ try{ return JSON.parse(localStorage.getItem(LS)) || {}; }catch(e){ return {}; } }
function lsWrite(o){ try{ localStorage.setItem(LS, JSON.stringify(o)); }catch(e){} }
let resolveReady; const ready = new Promise(r => resolveReady = r);

async function initStorage(){
  const local = lsRead();
  state = local.state || {}; myName = local.name || "";
  $("#myname").value = myName;
  paintAll();
  let db = null;
  try{ db = (window.claude && window.claude.use) ? await window.claude.use("db") : null; }catch(e){ db = null; }
  if(!db){ mode = "local"; resolveReady(); setStoreNote(); return; }
  dbApi = db; mode = "db"; resolveReady(); setStoreNote();
  const timer = setTimeout(()=>{ $("#store").textContent = "Saved data abhi load nahi hua, ho sakta hai purana dikh raha ho."; }, 10000);
  try{
    db.collection("outreach").onSnapshot(snap => {
      const next = {};
      snap.docs.forEach(d => { const v = d.data() || {}; next[d.id] = {status:v.status||"pending", note:v.note||"", sentAt:v.sentAt||""}; });
      state = next;
      if(!snap.metadata.fromCache){ clearTimeout(timer); setStoreNote(); }
      paintAll();
    }, () => { clearTimeout(timer); mode = "local"; setStoreNote(); });
    db.doc("settings/profile").onSnapshot(snap => {
      const v = snap.exists ? (snap.data()||{}) : {};
      if(typeof v.name === "string" && v.name !== myName && document.activeElement !== $("#myname")){
        myName = v.name; $("#myname").value = myName; paintDMs();
      }
    });
  }catch(e){ clearTimeout(timer); mode = "local"; setStoreNote(); }
}
function setStoreNote(){
  $("#store").textContent = mode === "db"
    ? "Progress is page par save hota hai, jis device se bhi kholein wahi dikhega."
    : "Progress sirf is browser mein save ho raha hai.";
}
const chain = {};
function persist(id){
  const s = state[id] || {status:"pending",note:"",sentAt:""};
  lsWrite({state, name:myName});
  chain[id] = (chain[id] || Promise.resolve()).then(async () => {
    await ready;
    if(mode !== "db" || !dbApi) return;
    try{ await dbApi.collection("outreach").doc(String(id)).set({status:s.status, note:s.note||"", sentAt:s.sentAt||"", updatedAt:new Date().toISOString()}); }
    catch(e){ toast("Save nahi hua, dobara try karein."); }
  });
}
let nameTimer;
function persistName(){
  lsWrite({state, name:myName});
  clearTimeout(nameTimer);
  nameTimer = setTimeout(async () => {
    await ready;
    if(mode !== "db" || !dbApi) return;
    try{ await dbApi.doc("settings/profile").set({name:myName}); }catch(e){}
  }, 600);
}

function get(id){ return state[id] || {status:"pending",note:"",sentAt:""}; }
function dmFor(c){ return c.dm.split("[Your Name]").join(myName.trim() || "[Your Name]"); }
function phoneInfo(c){
  if(!c.phone) return {type:"none"};
  const digits = c.phone.replace(/\D/g,"");
  const mobile = /^\+92 3\d\d /.test(c.phone);
  return {type: mobile ? "mobile" : "landline", digits};
}
function waLink(c){
  const p = phoneInfo(c);
  if(p.type === "none") return null;
  return "https://wa.me/" + p.digits + "?text=" + encodeURIComponent(dmFor(c));
}
function mapsLink(c){ return "https://www.google.com/maps/search/?api=1&query=" + encodeURIComponent(c.name + " " + c.city + " Pakistan"); }
function fmt(iso){ if(!iso) return ""; try{ return new Date(iso).toLocaleString("en-GB",{day:"numeric",month:"short",hour:"numeric",minute:"2-digit"}); }catch(e){ return ""; } }

function toast(msg){ const t = $("#toast"); t.textContent = msg; t.hidden = false; clearTimeout(toastT); toastT = setTimeout(()=>{ t.hidden = true; }, 2400); }

    const low = c.rating < 4;
    const phoneHtml = p.type === "none"

    const open = p.type === "none"
      ? '<a class="btn" href="'+esc(mapsLink(c))+'" target="_blank" rel="noopener" data-act="maps">Find on Google Maps</a>'
      : '<a class="btn primary" href="#" target="_blank" rel="noopener" data-act="open">Open WhatsApp</a>';
    return '<article class="row" id="r'+c.id+'" data-id="'+c.id+'" data-s="pending">' +
      '<div class="head"><span class="idx">'+String(c.id).padStart(2,"0")+'</span>' +

      '<div><span class="pill" data-pill data-s="pending">Not sent</span><div class="when" data-when></div></div></div>' +
      '<p class="dm clamp" data-dm></p>' +
      '<button class="linkbtn" data-act="toggle" type="button">Poora message dekhein</button>' +
      '<div class="actions">'+open+
      '<button class="btn" data-act="copy" type="button">Copy message</button>' +
      '<div class="seg" role="group" aria-label="Status">' + ST.map(s => '<button type="button" data-st="'+s.k+'" aria-pressed="false">'+s.label+'</button>').join("") + '</div></div>' +
      '<p class="hint" data-hint hidden>WhatsApp khul gaya? Message bhejne ke baad "Sent" dabayein taake yahan nishan lag jaye.</p>' +
      '<input class="note" data-note type="text" placeholder="Note (reply, kab dobara message karna hai...)" aria-label="Note">' +
      '</article>';
  }).join("");
}

function paintDMs(){
  CLINICS.forEach(c => {
    const row = $("#r"+c.id);
    row.querySelector("[data-dm]").textContent = dmFor(c);
    const a = row.querySelector('[data-act="open"]'); if(a) a.href = waLink(c);
  });
  const empty = !myName.trim();
  $("#myname").classList.toggle("need", empty);
  $("#nameHelp").textContent = empty
    ? "Pehle apna naam likhein, har DM mein [Your Name] ki jagah lag jayega."
    : "Naam har DM mein lag gaya hai.";
  paintNext();
}
function paintRow(c){
  const row = $("#r"+c.id), s = get(c.id);
  row.dataset.s = s.status;
  const pill = row.querySelector("[data-pill]"); pill.dataset.s = s.status; pill.textContent = LABEL[s.status];
  row.querySelector("[data-when]").textContent = s.status !== "pending" && s.sentAt ? "Sent " + fmt(s.sentAt) : "";
  row.querySelectorAll(".seg button").forEach(b => b.setAttribute("aria-pressed", String(b.dataset.st === s.status)));
  const n = row.querySelector("[data-note]"); if(document.activeElement !== n) n.value = s.note || "";
  if(s.status !== "pending") row.classList.remove("opened");
  row.querySelector("[data-hint]").hidden = !(row.classList.contains("opened") && s.status === "pending");
}
function counts(){
  const k = {pending:0,sent:0,replied:0,follow:0,no:0};
  CLINICS.forEach(c => k[get(c.id).status]++);
  return k;
}
function paintStats(){
  const k = counts(), total = CLINICS.length, done = total - k.pending;
  $("#cnt").textContent = done; $("#of").textContent = "of " + total + " contacted";
  const colors = {sent:"var(--c-sent)",replied:"var(--c-replied)",follow:"var(--c-follow)",no:"var(--c-no)"};
  $("#bar").innerHTML = ["replied","sent","follow","no"].map(x => '<i style="width:'+(k[x]/total*100)+'%;background:'+colors[x]+'"></i>').join("");
  $("#bar").setAttribute("aria-label", done+" of "+total+" contacted");
  $("#legend").innerHTML = ["pending","sent","replied","follow","no"].map(x =>
    '<span><i class="dot" style="background:'+(colors[x]||"var(--c-pending)")+'"></i>'+LABEL[x]+' '+k[x]+'</span>').join("");
  const chips = [["all","All",total],["pending","Not sent",k.pending],["sent","Sent",k.sent],["replied","Replied",k.replied],["follow","Follow up",k.follow],["no","Not interested",k.no]];
  $("#chips").innerHTML = chips.map(c => '<button class="chip" type="button" data-f="'+c[0]+'" aria-pressed="'+(fStatus===c[0])+'">'+c[1]+' '+c[2]+'</button>').join("");
}

  const p = phoneInfo(c);

    '<div class="grp"><a class="btn" href="'+esc(waLink(c))+'" target="_blank" rel="noopener" data-next-open="'+c.id+'">Open WhatsApp</a><button class="btn ghost" type="button" data-jump="'+c.id+'">Row par jayein</button></div>';
}
function applyFilter(){
  let shown = 0;
  CLINICS.forEach(c => {
    const s = get(c.id).status;
    const ok = (fStatus === "all" || s === fStatus) && (fCity === "all" || c.city === fCity) && (!fLow || c.rating < 4) &&
      (!fQ || (c.name + " " + c.city).toLowerCase().includes(fQ));
    $("#r"+c.id).hidden = !ok; if(ok) shown++;
  });
  $("#empty").hidden = shown > 0;
}
function paintAll(){

}

function setStatus(id, status){
  const cur = get(id);
  const next = cur.status === status ? "pending" : status;
  state[id] = {status:next, note:cur.note||"", sentAt: next === "pending" ? "" : (cur.sentAt || new Date().toISOString())};
  persist(id); paintRow(CLINICS.find(c=>c.id===id)); paintStats(); paintNext(); applyFilter();
}
async function copyText(text){
  try{ await navigator.clipboard.writeText(text); return true; }catch(e){}
  try{
    const ta = document.createElement("textarea"); ta.value = text; ta.style.position="fixed"; ta.style.opacity="0";
    document.body.appendChild(ta); ta.select(); const ok = document.execCommand("copy"); ta.remove(); return ok;
  }catch(e){ return false; }
}
